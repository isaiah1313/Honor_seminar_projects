\section{特殊相対論の復習と拡張の必要性}

\begin{frame}{特殊相対論の復習}
    $\blacksquare$ 慣性系と時空の表現
    \begin{itemize}
        \item 慣性の法則が成立する系で物理現象を記述する
        \item 時空の点は4元位置ベクトル $x^\mu = [ct, x, y, z]^T$ で指定する
    \end{itemize}

    \vspace{0.8em}
    $\blacksquare$ 慣性系間の変換（ローレンツ変換）
    \begin{itemize}
        \item 異なる慣性系への座標変換は線形変換となる:
        \begin{equation*}
            x'^\mu = \tensor{\Lambda}{^\mu_\nu} x^\nu
        \end{equation*}
        \item この変換のもとで世界間隔 $ds^2$ は不変に保たれる:
        \begin{equation*}
            ds^2 = \eta_{\mu\nu} dx^\mu dx^\nu = \mathrm{const.} \quad (\eta_{\mu\nu} = \mathrm{diag}(1,-1,-1,-1))
        \end{equation*}
    \end{itemize}
\end{frame}



\begin{frame}{特殊相対論の拡張1:座標変換}
    \begin{block}{特殊相対論で用いる座標}
        直交する4方向の直線を使って時空の点を表現する.

        座標変換は慣性系同士のものに限られている.
    \end{block}

    $\Rightarrow$もっと自由に曲線座標を含めた座標変換をしたい!

    例:
    \begin{align*}
        \left\{
        \begin{aligned}
        x' &= x - \frac{1}{2}at^2\\
        t' &= t
        \end{aligned}
        \right.
        \qquad\qquad
        \left\{
        \begin{aligned}
            r &= \sqrt{x^2 + y^2 + z^2} \\
            \theta &= \arccos\left(\frac{z}{r}\right) \\
            \phi &= \arctan\left(\frac{y}{x}\right)\\
            t' &= t
        \end{aligned}
        \right.
    \end{align*}
\end{frame}

\begin{frame}{特殊相対論の拡張2: 曲がっている空間への対応}
    $\blacksquare$ 一様な重力場と座標系
    \begin{itemize}
        \item 一様な重力 $\bm{g}$ が働く系における運動方程式:
        \begin{equation*}
            m \frac{d^2\bm{x}}{dt^2} = \bm{F} + \textcolor{red}{m\bm{g}}

        \begin{equation*}
            m \frac{d^2\bm{x}'}{dt^2} = \bm{F} \quad \text{（重力項が消去される）}
        \end{equation*}
    \end{itemize}

\end{frame}

\section{一般相対論の舞台}

\begin{frame}{一般相対論の舞台:局所座標}
    \begin{block}{曲がっている時空の表現方法}
        曲がっている時空の点の位置は4個の実数値で指定できる.

        また，点の位置を指定するのに使う実数値の組を\textcolor{red}{局所座標}と呼ぶ.
    \end{block}
    例:

    球面$x^2+y^2+z^2=1$の$z>0$半球上のの点は
    
    緯度と経度を用いて$(\theta,\phi)$で指定することもできるし，
    
    $xy$平面に射影した座標$(x,y)$で指定することもできる.

\end{frame}

\begin{frame}{一般相対論の舞台:テンソル}
    \begin{block}{時空上の物理量}
        物理量は座標変換 $(x^\mu \to y^\mu)$ に対する変換則で分類される.
        
        \vspace{0.4em}
        \centering
        \renewcommand{\arraystretch}{1.5}
        \small
        \begin{tabular}{lll}
            \hline
            分類 & 記法 & 変換則 \\
            \hline
            スカラー & $\phi$ & $\phi' = \phi$ \\
            反変ベクトル & $A^\mu$ & $A'^\mu = \pdiff{y^\mu}{x^\nu} A^\nu$ \\
            共変ベクトル & $A_\mu$ & $A'_\mu = \pdiff{x^\nu}{y^\mu} A_\nu$ \\
            2階テンソル & $T^{\mu\nu}$ & $T'^{\mu\nu} = \pdiff{y^\mu}{x^\rho} \pdiff{y^\nu}{x^\sigma} T^{\rho\sigma}$ \\
            \hline
        \end{tabular}
        \vspace{0.3em}
    \end{block}
    \vspace{0.3em}
    $\Rightarrow$ 一般のテンソルは、上付き添字1つにつき $\pdiff{y}{x}$、下付き添字1つにつき $\pdiff{x}{y}$ が掛かる.
\end{frame}


\begin{frame}{一般相対論の舞台:計量}
    \begin{block}{時空の2点間の距離の測り方}
        局所座標として$x^\mu$を採用する.

        点P$(x)$と点Q$(x + dx)$の距離の二乗$ds^2$は一般に$dx^\mu$の二次形式
        \begin{equation*}
            ds^2 = g_{\mu_\nu} dx^\mu dx^\nu
        \end{equation*}
        で計算できる.この係数行列$g_{\mu_\nu}$を\textcolor{red}{計量テンソル}と呼ぶ.

        上付きの計量テンソル$g^{\mu_\nu}$を$g_{\mu\nu}$の逆行列として定義する:\>$g_{\mu\nu}g^{\nu\lambda}=\delta^\lambda_\mu$

        これらの計量テンソルを用いてベクトルやテンソルの添字を上下する:
        \begin{align*}
            A^\mu = g^{\mu\nu}A_\nu \quad,\quad A_\mu = g_{\mu\nu}A^\nu
        \end{align*}
    \end{block}
\end{frame}


\begin{frame}{一般相対論の舞台:ガリレイ的座標}
    \begin{block}{物理的な要請}
        宇宙のどこにいても，自由落下する観測者の近くでは慣性系になっている.
        \begin{center}
            $\Downarrow$
        \end{center}
        時空の各点で局所的に計量が$g_{\mu\nu} = \eta_{\mu\nu}$となるガリレイ的な座標$\xi^\mu$を選択できる.
    \end{block}
    一般の曲線座標への変換$\xi^\mu\to x^\mu$の元で，計量は
    \begin{equation*}
        g_{\mu\nu} = \eta_{\rho\sigma}\pdiff{\xi^\rho}{x^\mu}\pdiff{\xi^\sigma}{x^\nu}
    \end{equation*}
    のように合同変換される.
\end{frame}

\begin{frame}{一般相対論の舞台:積分要素}
    \begin{block}{4次元体積要素とヤコビアンの同定}
        ガリレイ的座標での体積要素 $d\Omega = d^4\xi$ を一般座標 $x^\mu$ へ変数変換すると、
        \begin{equation*}
            d\Omega = \left| \det\left(\pdiff{\xi}{x}\right) \right| d^4x \quad (d^4x \equiv dx^0 dx^1 dx^2 dx^3)
        \end{equation*}
        計量の変換式 $g_{\mu\nu} = \eta_{\rho\sigma} \pdiff{\xi^\rho}{x^\mu} \pdiff{\xi^\sigma}{x^\nu}$ の行列式をとると（$\det\eta = -1$ より）、
        \begin{equation*}
            g = - \left| \det\left(\pdiff{\xi}{x}\right) \right|^2
            \quad \Longrightarrow \quad
            \left| \det\left(\pdiff{\xi}{x}\right) \right| = \sqrt{-g}
        \end{equation*}
    \end{block}

    \vspace{0.6em}
    \centering
    \large
    $\therefore \quad \bm{d\Omega = \sqrt{-g} \, d^4x}$ \normalsize \quad （不変体積要素）
\end{frame}

\begin{frame}{一般相対論の舞台:曲がった時空での微分}
    $\blacksquare$ スカラー量$\phi$の微分は共変ベクトルとしての変換則を満たす:
    \begin{equation*}
        \pdiff{\phi}{x^\mu} = \pdiff{\phi}{y^\nu}\pdiff{y^\nu}{x^\mu}
    \end{equation*}
    $\blacksquare$ ベクトル量$A^\mu$の微分はベクトルとしての変換則を満たさない:
    \begin{equation*}
        {A'}^\mu = \pdiff{y^\mu}{x^\nu}A^\nu \quad\text{より，}\quad d{A'}^\mu = \pdiff{y^\mu}{x^\nu}dA^\nu + A^\nu\frac{\partial^2 y^\mu}{\partial x^\nu \partial x^\lambda}dx^\lambda
    \end{equation*}
    $\Rightarrow$ ベクトルの偏微分は座標系に依存する量であり，物理を記述するには不満足である(テンソルになっていない).

        ベクトルの偏微分がうまくいかなかった理由:

    曲がっている空間の異なる2点のベクトルの引き算をしたから!!
\end{frame}


\begin{frame}{一般相対論の舞台: ベクトルの平行移動と微分}
    \begin{block}{ベクトルの平行移動と差分の計算}
        点 $\mathrm{P}(x)$ のベクトル $A^\mu(x)$ を点 $\mathrm{Q}(x+dx)$ へ平行移動させたベクトル:
        \begin{equation*}
            A_\parallel^\mu(x+dx) = A^\mu(x) - \Gamma^\mu_{\nu\lambda}(x) A^\nu(x) dx^\lambda
        \end{equation*}
        同じ点 $\mathrm{Q}$ でのベクトル $A^\mu(x+dx)$ との差分を計算すると、
        \begin{align*}
            \Delta A^\mu 
            &= A^\mu(x+dx) - A_\parallel^\mu(x+dx) \\
            &= \left( \pdiff{A^\mu(x)}{x^\lambda} + \Gamma^\mu_{\nu\lambda}(x) A^\nu(x) \right) dx^\lambda \;\equiv\; (\nabla_\lambda A^\mu) dx^\lambda
        \end{align*}
    \end{block}

    \vspace{0.4em}
    \centering
    $\Longrightarrow \quad \displaystyle \bm{\nabla_\lambda A^\mu \equiv \pdiff{A^\mu(x)}{x^\lambda} + \Gamma^\mu_{\nu\lambda}(x) A^\nu(x)}$ \quad （\textbf{共変微分}）
\end{frame}

\begin{frame}{一般相対論の舞台: 共変ベクトルとテンソルの共変微分}
    $\blacksquare$ 共変ベクトルの共変微分
    \begin{itemize}
        \item スカラーの内積 $A_\mu B^\mu$ の微分が通常の偏微分に一致する要請から定まる:
        \begin{equation*}
            \nabla_\lambda(A_\mu B^\mu) = \partial_\lambda(A_\mu B^\mu)
        \end{equation*}
        \item 反変の公式 $\nabla_\lambda B^\mu = \partial_\lambda B^\mu + \Gamma^\mu_{\nu\lambda} B^\nu$ とライプニッツ則を用いると:
        \begin{equation*}
            \nabla_\lambda A_\mu = \pdiff{A_\mu}{x^\lambda} - \Gamma^\nu_{\mu\lambda} A_\nu \quad \text{（接続項の符号が反転する）}
        \end{equation*}
    \end{itemize}

    \vspace{0.4em}
    $\blacksquare$ 一般のテンソルへの拡張
    \begin{itemize}
        \item 各添字に対して「上付き添字には $+\Gamma$」「下付き添字には $-\Gamma$」が作用する
        \item 例: 2階共変テンソル $T_{\mu\nu}$（計量 $g_{\mu\nu}$ など）の共変微分
        \begin{equation*}
            \nabla_\lambda T_{\mu\nu} = \pdiff{T_{\mu\nu}}{x^\lambda} - \Gamma^\sigma_{\mu\lambda} T_{\sigma\nu} - \Gamma^\sigma_{\nu\lambda} T_{\mu\sigma}
        \end{equation*}
    \end{itemize}
\end{frame}

\begin{frame}{計量によるクリストッフェル記号の決定}
    $\blacksquare$ 平行移動に対する要請
    \begin{itemize}
        \item ベクトルのノルム保存（計量条件）: $\nabla_\lambda g_{\mu\nu} = 0$
        \item 下付き添字の対称性(ガリレイ的座標の存在): $\Gamma^\mu_{\nu\lambda} = \Gamma^\mu_{\lambda\nu}$
    \end{itemize}

    \vspace{0.8em}
    $\blacksquare$ クリストッフェル記号の決定
    \begin{itemize}
        \item 計量条件 $\partial_\lambda g_{\mu\nu} = \Gamma^\sigma_{\mu\lambda} g_{\sigma\nu} + \Gamma^\sigma_{\nu\lambda} g_{\mu\sigma}$ と添字の対称性から連立方程式を解くことで、計量とその偏微分により一意に定まる:
    \end{itemize}

    \begin{equation*}
        \Gamma^\mu_{\nu\lambda} = \frac{1}{2} g^{\mu\rho} \left( \pdiff{g_{\rho\nu}}{x^\lambda} + \pdiff{g_{\rho\lambda}}{x^\nu} - \pdiff{g_{\nu\lambda}}{x^\rho} \right)
    \end{equation*}
\end{frame}

\begin{frame}{一般相対論の舞台: リーマン曲率テンソル}
    $\blacksquare$ 微小平行四辺形による曲率の定義
    \begin{itemize}
        \item 点 $\mathrm{P}(x)$ から点 $\mathrm{S}(x+du+dv)$ へ至る2つの経路を考える:
        \begin{itemize}
            \item 経路1: $\mathrm{P} \to \mathrm{Q}(x+du) \to \mathrm{S}(x+du+dv)$
            \item 経路2: $\mathrm{P} \to \mathrm{R}(x+dv) \to \mathrm{S}(x+du+dv)$
        \end{itemize}
        \item 各経路でベクトル $A^\mu$ を平行移動させた結果の差を計算する:
        \begin{equation*}
            \Delta A^\mu = A^\mu_{\mathrm{P \to Q \to S}} - A^\mu_{\mathrm{P \to R \to S}} = R^\mu_{\ \nu\rho\sigma} A^\nu(x) du^\rho dv^\sigma
        \end{equation*}
    \end{itemize}
    ただし，$R^\mu_{\ \nu\rho\sigma}$の具体的な表式は以下である:
    \begin{equation*}
        R^\mu_{\ \nu\rho\sigma} = \pdiff{\Gamma^\mu_{\nu\sigma}}{x^\rho} - \pdiff{\Gamma^\mu_{\nu\rho}}{x^\sigma} + \Gamma^\mu_{\lambda\rho}\Gamma^\lambda_{\nu\sigma} - \Gamma^\mu_{\lambda\sigma}\Gamma^\lambda_{\nu\rho}
    \end{equation*}
\end{frame}

\begin{frame}{一般相対論の舞台: 曲率テンソルの自由度と曲率を表す量}
    $\blacksquare$ 代称性と独立成分数（自由度 $256 \to 20$）
    \begin{itemize}\small
        \item 4階共変成分 $R_{\mu\nu\rho\sigma} = g_{\mu\lambda} R^\lambda_{\ \nu\rho\sigma}$ は以下の関係式を満たす:
        \begin{itemize}\footnotesize
            \item 前後ペアの反対称性: $R_{\mu\nu\rho\sigma} = -R_{\nu\mu\rho\sigma} = -R_{\mu\nu\sigma\rho}$
            \item ペアの入れ替え対称性: $R_{\mu\nu\rho\sigma} = R_{\rho\sigma\mu\nu}$
            \item 第1ビアンキ恒等式: $R_{\mu\nu\rho\sigma} + R_{\mu\rho\sigma\nu} + R_{\mu\sigma\nu\rho} = 0$
        \end{itemize}
        \item 4次元時空において、独立成分の数は $4^4=256$ 個から 20 個に減少する
    \end{itemize}

    \vspace{0.2em}
    $\blacksquare$ 縮約によって得られる曲率の量
    \begin{itemize}\small
        \item リッチテンソル（2階対称テンソル、独立成分 10 個）:
        \begin{equation*}
            R_{\mu\nu} \equiv R^\lambda_{\ \mu\lambda\nu} \quad (R_{\mu\nu} = R_{\nu\mu})
        \end{equation*}
        \item スカラー曲率（リッチスカラー）:
        \begin{equation*}
            R \equiv g^{\mu\nu} R_{\mu\nu}
        \end{equation*}
    \end{itemize}
\end{frame}

\section{最小作用の原理}

\begin{frame}{最小作用の原理とは}
    $\blacksquare$ 時空全体における作用の定義
    \begin{itemize}
        \item 系の状態$g_{\mu\nu}$から定まるスカラー量 $\mathcal{L}$ を時空領域全体で積分した量:
        \begin{equation*}
            S = \int \mathcal{L} \sqrt{-g(x)} \, d^4x
        \end{equation*}
    \end{itemize}

    \vspace{0.4em}
    $\blacksquare$ 場の変分と停留条件
    \begin{itemize}
        \item 時空の境界 $\partial \mathcal{M}$ で消える仮想的な微小変化$g_{\mu\nu}\to g_{\mu\nu}+\delta g_{\mu\nu}$を考える:
        \item 物理的に実現する場の配位は、任意の微小変化に対して作用が停留（不変）となる:
        \begin{equation*}
            \delta S = 0
        \end{equation*}
    \end{itemize}
\end{frame}

\section{アインシュタイン方程式}

\begin{frame}{時空を決定する作用}
    $\blacksquare$ 作用に課す条件
    \begin{itemize}
        \item 座標変換に対して不変
        \item 境界項を除き，$g_{\mu\nu}$ の1階微分までを含む
    \end{itemize}

    \vspace{0.6em}
    $\blacksquare$ リッチスカラー $R$ はこの条件を満たす
    \begin{itemize}
        \item $R$ はスカラーなので，$\int R\sqrt{-g}\,d^4x$ は不変
        \item 2階微分を含む部分は，境界項として分離できる
    \end{itemize}

    \vspace{0.4em}
    \begin{center}
        $\displaystyle \boxed{\mathcal{L}=R}$
        \qquad {\small （比例定数・宇宙定数は省略）}
    \end{center}
    {\footnotesize 境界付近で変分をゼロとすれば，境界項は変分に寄与しない.}
    % Reference: David Tong, General Relativity, section 4.1.
    % https://www.damtp.cam.ac.uk/user/tong/gr/grhtml/S4.html
\end{frame}

\begin{frame}{時空の運動方程式}
    $\blacksquare$ 重力と物質を合わせた作用
    \begin{equation*}
        S = -\frac{c^3}{16\pi G}\int R\sqrt{-g}\,d^4x + S_{\mathrm m}
        \qquad \text{（$S_{\mathrm m}$: 物質の作用）}
    \end{equation*}

    $\blacksquare$ 物質の作用を計量で変分し，$T_{\mu\nu}$ を定義
    \begin{equation*}
        \delta S_{\mathrm m}
        = \frac{1}{2c}\int
        \underbrace{T_{\mu\nu}}_{\text{エネルギー運動量テンソル}}
        \,\delta g^{\mu\nu}\sqrt{-g}\,d^4x
    \end{equation*}

    $\blacksquare$ 任意の計量の変分に対して $\delta S=0$
    \begin{equation*}
        \boxed{R_{\mu\nu}-\frac{1}{2}g_{\mu\nu}R
        = \frac{8\pi G}{c^4}T_{\mu\nu}}
        \qquad \text{（アインシュタイン方程式）}
    \end{equation*}
    % Signature (+---), x^0=ct, and the curvature convention defined earlier.
    % With physical matter actions, delta S_m = +(1/2c) int sqrt(-g) T_mn delta g^mn.
    % Boundary variations vanish as on the preceding slide; Lambda is omitted.
    % Reference: David Tong, General Relativity, section 4.5 (adapted signs and c).
    % https://www.damtp.cam.ac.uk/user/tong/gr/grhtml/S4.html
\end{frame}
